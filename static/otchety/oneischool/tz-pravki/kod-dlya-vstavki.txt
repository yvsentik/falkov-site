================================================================

1. Старый поддомен (пункт 4 ТЗ). Конфигурация nginx
----------------------------------------------------
server {
    listen 443 ssl;
    server_name old.oneischool.com;
    # сертификаты оставить прежние
    return 301 https://oneischool.com$request_uri;
}
server {
    listen 80;
    server_name old.oneischool.com;
    return 301 https://oneischool.com$request_uri;
}
Проверка: https://old.oneischool.com/chastnye-shkoly/ должен отдавать код 301 и адрес https://oneischool.com/chastnye-shkoly/

2. Карточки педагогов и событий (пункт 5 ТЗ). functions.php темы
-----------------------------------------------------------------
// убрать педагогов, события и пользователей из карты сайта
add_filter('wp_sitemaps_post_types', function ($types) {
    unset($types['pedagog'], $types['event']);
    return $types;
});
add_filter('wp_sitemaps_add_provider', function ($provider, $name) {
    return $name === 'users' ? false : $provider;
}, 10, 2);

// noindex, follow вместо noindex, nofollow
add_filter('wp_robots', function ($robots) {
    if (is_singular(['pedagog', 'event'])) {
        $robots['noindex'] = true;
        $robots['follow']  = true;
        unset($robots['nofollow'], $robots['index']);
    }
    return $robots;
});
Проверка: в https://oneischool.com/wp-sitemap.xml не должно остаться строк posts-pedagog, posts-event и users.

3. Canonical (пункт 8 ТЗ). functions.php темы
----------------------------------------------
// страницы пагинации блога ссылаются сами на себя
add_filter('get_canonical_url', function ($url) {
    if (is_paged()) {
        return get_pagenum_link(get_query_var('paged'));
    }
    return $url;
});
// фильтр ?category= закрыть от индексации
add_filter('wp_robots', function ($robots) {
    if (isset($_GET['category'])) {
        $robots['noindex'] = true;
        $robots['follow']  = true;
    }
    return $robots;
});
Если canonical выводится в шаблоне темы вручную, поправить там же: на /blog/page/N/ ставить адрес самой страницы.
Добавить canonical на: /dopolnitelnoe-obrazovanie/, /teachers/, /blog/category/stati/,
/blog/category/novosti-horoshevskiy/, /blog/category/novosti-mosfilmovskiy/
